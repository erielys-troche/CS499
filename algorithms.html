Algorithms and data structures: Course Planner

A C++ program from CS 300 that loads courses and prerequisites, now with a self-balancing tree and a prerequisite graph.

Original artifact

The CS 300 project reads a course file into a hash table and a binary search tree, then prints courses in order. A plain binary search tree is fast only if the data arrives in a random order. Sorted input makes it a chain.

What I changed

  • AVL tree: after each insert it checks the balance factor and fixes it with rotations (LL, RR, LR, RL). Nine sorted courses give a height of 4 instead of 9.
  • Prerequisite graph: edges point from a prerequisite to the course that needs it. Kahn's algorithm gives a valid order and fails when there's a cycle, so the program prints "circular prerequisite dependency detected."
  • Memory cleanup: while reviewing my own work I found the AVL tree never freed its nodes. I added a destructor, and Valgrind went from 1,410 bytes still in use at exit to 0.

BinarySearchTree.hpp
// BinarySearchTree.hpp
// Original artifact from CS 300 (Course Planner).
//
// NOTE (flagged in the Milestone One code review): this is a plain,
// unbalanced binary search tree. If courses load in sorted order (which
// they often do, since course IDs are naturally sorted in most catalog
// files), this tree degenerates into a straight line and search/insert
// degrade from O(log n) to O(n). This is addressed in the Milestone
// Three enhancement by converting this into a self-balancing AVL tree.
#ifndef BINARY_SEARCH_TREE_HPP
#define BINARY_SEARCH_TREE_HPP

#include "Course.hpp"
#include <iostream>

class BinarySearchTree {
private:
    struct Node {
        Course course;
        Node* left;
        Node* right;
        explicit Node(Course c) : course(c), left(nullptr), right(nullptr) {}
    };

    Node* root;

    void insertRecursive(Node* node, Course course) {
        if (course.courseId < node->course.courseId) {
            if (node->left == nullptr) {
                node->left = new Node(course);
            } else {
                insertRecursive(node->left, course);
            }
        } else {
            if (node->right == nullptr) {
                node->right = new Node(course);
            } else {
                insertRecursive(node->right, course);
            }
        }
    }

    void printInOrderHelper(Node* node) const {
        if (node != nullptr) {
            printInOrderHelper(node->left);
            std::cout << node->course.courseId << " - " << node->course.courseName << std::endl;
            printInOrderHelper(node->right);
        }
    }

    Node* findNode(Node* node, const std::string& courseId) const {
        if (node == nullptr || node->course.courseId == courseId) {
            return node;
        }
        if (courseId < node->course.courseId) {
            return findNode(node->left, courseId);
        }
        return findNode(node->right, courseId);
    }

    int heightHelper(Node* node) const {
        if (node == nullptr) return 0;
        int leftHeight = heightHelper(node->left);
        int rightHeight = heightHelper(node->right);
        return 1 + (leftHeight > rightHeight ? leftHeight : rightHeight);
    }

public:
    BinarySearchTree() : root(nullptr) {}

    void insert(Course course) {
        if (root == nullptr) {
            root = new Node(course);
        } else {
            insertRecursive(root, course);
        }
    }

    void printInOrder() const {
        printInOrderHelper(root);
    }

    bool contains(const std::string& courseId) const {
        return findNode(root, courseId) != nullptr;
    }

    int getHeight() const {
        return heightHelper(root);
    }
};

#endif

// HashTable.hpp
// Original artifact from CS 300 (Course Planner).
//
// Basic chaining hash table. Table size is fixed and never resizes, so
// collisions increase (and performance degrades) as more courses are
// added -- this was noted as a limitation in the Milestone One review.
#ifndef HASH_TABLE_HPP
#define HASH_TABLE_HPP

#include "Course.hpp"
#include <vector>
#include <list>
#include <iostream>

class HashTable {
private:
    static const int TABLE_SIZE = 10;
    std::vector<std::list<Course>> table;

    unsigned int hash(const std::string& courseId) const {
        unsigned int sum = 0;
        for (char c : courseId) {
            sum += static_cast<unsigned int>(c);
        }
        return sum % TABLE_SIZE;
    }

public:
    HashTable() : table(TABLE_SIZE) {}

    void insert(Course course) {
        unsigned int index = hash(course.courseId);
        table[index].push_back(course);
    }

    void printAll() const {
        for (const auto& bucket : table) {
            for (const auto& course : bucket) {
                std::cout << course.courseId << " - " << course.courseName << std::endl;
            }
        }
    }

    bool contains(const std::string& courseId) const {
        unsigned int index = hash(courseId);
        for (const auto& course : table[index]) {
            if (course.courseId == courseId) return true;
        }
        return false;
    }
};

#endif

// main.cpp
// Original artifact from CS 300 (Course Planner).
//
// NOTE (flagged in the Milestone One review): validateCourses() only

// list -- it cannot detect a circular prerequisite chain (e.g. Course A

void loadCourses(const std::string& filename) {
    std::ifstream file(filename);
    if (!file.is_open()) {
        std::cout << "Error: File not found." << std::endl;
        return;
    }

    std::string line;
    while (std::getline(file, line)) {
        std::stringstream ss(line);
        std::string token;
        std::vector<std::string> tokens;
        while (std::getline(ss, token, ',')) {
            tokens.push_back(token);
        }
        if (tokens.size() < 2) continue;

        Course course;
        course.courseId = tokens[0];
        course.courseName = tokens[1];
        for (size_t i = 2; i < tokens.size(); i++) {
            course.prerequisites.push_back(tokens[i]);
        }

        courseVector.push_back(course);
        courseTree.insert(course);
        courseHashTable.insert(course);
    }

    std::cout << "Courses loaded successfully." << std::endl;
    std::cout << "(BST height after loading: " << courseTree.getHeight()
              << " for " << courseVector.size() << " courses)" << std::endl;
}

    for (const auto& course : courseVector) {
        for (const auto& prereq : course.prerequisites) {
            bool found = false;
            for (const auto& c : courseVector) {
                if (c.courseId == prereq) {
                    found = true;
                    break;
                }

            if (!found) {
                std::cout << "ERROR: Prerequisite " << prereq
                          << " missing for " << course.courseId << std::endl;
                return false;
            }
        }
    }
    std::cout << "All prerequisites are valid." << std::endl;
    return true;
}

        if (course.courseId == courseId) {

            if (course.prerequisites.empty()) {
                std::cout << "None";
            }
            for (size_t i = 0; i < course.prerequisites.size(); i++) {
                std::cout << course.prerequisites[i];
                if (i != course.prerequisites.size() - 1) std::cout << ", ";
            }
            std::cout << std::endl;
            return;
        }

    std::cout << "Error: Course not found." << std::endl;
}

}

AVLTree.hpp
// AVLTree.hpp
// ENHANCEMENT (Milestone Three — Algorithms and Data Structures).
//
// Replaces the original plain, unbalanced BinarySearchTree. This tree
// tracks a height at every node and performs rotations on insert to
// keep itself balanced, guaranteeing O(log n) search and insert no
// matter what order courses are loaded in -- including the sorted-ID
// worst case that broke the original implementation.
#ifndef AVL_TREE_HPP
#define AVL_TREE_HPP

#include "Course.hpp"
#include <algorithm>
#include <iostream>

class AVLTree {
private:
    struct Node {
        Course course;
        Node* left;
        Node* right;
        int height;
        explicit Node(Course c) : course(c), left(nullptr), right(nullptr), height(1) {}
    };

    Node* root;

    // Frees every node in the tree (children first, then the parent).
    void destroy(Node* node) {
        if (node == nullptr) return;
        destroy(node->left);
        destroy(node->right);
        delete node;
    }

    int height(Node* node) const {
        return node == nullptr ? 0 : node->height;
    }

    int balanceFactor(Node* node) const {
        return node == nullptr ? 0 : height(node->left) - height(node->right);
    }

    void updateHeight(Node* node) {
        node->height = 1 + std::max(height(node->left), height(node->right));
    }

    Node* rotateRight(Node* y) {
        Node* x = y->left;
        Node* t2 = x->right;
        x->right = y;
        y->left = t2;
        updateHeight(y);
        updateHeight(x);
        return x;
    }

    Node* rotateLeft(Node* x) {
        Node* y = x->right;
        Node* t2 = y->left;
        y->left = x;
        x->right = t2;
        updateHeight(x);
        updateHeight(y);
        return y;
    }

    Node* insertRecursive(Node* node, Course course) {
        if (node == nullptr) {
            return new Node(course);
        }
        if (course.courseId < node->course.courseId) {
            node->left = insertRecursive(node->left, course);
        } else {
            node->right = insertRecursive(node->right, course);
        }

        updateHeight(node);
        int balance = balanceFactor(node);

        // Left-Left case
        if (balance > 1 && node->left != nullptr && course.courseId < node->left->course.courseId) {
            return rotateRight(node);
        }
        // Right-Right case
        if (balance < -1 && node->right != nullptr && course.courseId > node->right->course.courseId) {
            return rotateLeft(node);
        }
        // Left-Right case
        if (balance > 1 && node->left != nullptr && course.courseId > node->left->course.courseId) {
            node->left = rotateLeft(node->left);
            return rotateRight(node);
        }
        // Right-Left case
        if (balance < -1 && node->right != nullptr && course.courseId < node->right->course.courseId) {
            node->right = rotateRight(node->right);
            return rotateLeft(node);
        }

        return node;
    }

    void printInOrderHelper(Node* node) const {
        if (node != nullptr) {
            printInOrderHelper(node->left);
            std::cout << node->course.courseId << " - " << node->course.courseName << std::endl;
            printInOrderHelper(node->right);
        }
    }

    Node* findNode(Node* node, const std::string& courseId) const {
        if (node == nullptr || node->course.courseId == courseId) {
            return node;
        }
        if (courseId < node->course.courseId) {
            return findNode(node->left, courseId);
        }
        return findNode(node->right, courseId);
    }

public:
    AVLTree() : root(nullptr) {}
    ~AVLTree() { destroy(root); }

    // A copy would share the same nodes and free them twice, so copying is
    // turned off. (The program only ever uses one tree.)
    AVLTree(const AVLTree&) = delete;
    AVLTree& operator=(const AVLTree&) = delete;

    void insert(Course course) {
        root = insertRecursive(root, course);
    }

    void printInOrder() const {
        printInOrderHelper(root);
    }

    bool contains(const std::string& courseId) const {
        return findNode(root, courseId) != nullptr;
    }

    int getHeight() const {
        return height(root);
    }
};

#endif

CourseGraph.hpp
// CourseGraph.hpp
// ENHANCEMENT (Milestone Three — Algorithms and Data Structures).
//
// Models course prerequisites as a directed graph (an edge from a
// prerequisite to the course that requires it) and implements Kahn's
// algorithm for topological sort. This is new functionality that did
// not exist in the original artifact:
//   1. It can detect a circular prerequisite dependency (e.g. Course A
//      requires Course B, and Course B requires Course A), which the
//      original validateCourses() could not do.
//   2. It generates a valid course-taking order for students/advisors.
#ifndef COURSE_GRAPH_HPP
#define COURSE_GRAPH_HPP

#include <string>
#include <vector>
#include <unordered_map>
#include <queue>
#include <iostream>

class CourseGraph {
private:
    std::unordered_map<std::string, std::vector<std::string>> adjacency;
    std::unordered_map<std::string, int> inDegree;

public:
    void addCourse(const std::string& courseId) {
        if (adjacency.find(courseId) == adjacency.end()) {
            adjacency[courseId] = {};
            inDegree[courseId] = 0;
        }
    }

    void addPrerequisiteEdge(const std::string& prereqId, const std::string& courseId) {
        addCourse(prereqId);
        addCourse(courseId);
        adjacency[prereqId].push_back(courseId);
        inDegree[courseId]++;
    }

    // Returns a valid course order using Kahn's algorithm, or an empty
    // vector (with an explanatory message) if a cycle is detected.
    std::vector<std::string> topologicalSort() const {
        std::unordered_map<std::string, int> localInDegree = inDegree;
        std::queue<std::string> ready;

        for (const auto& pair : localInDegree) {
            if (pair.second == 0) {
                ready.push(pair.first);
            }
        }

        std::vector<std::string> order;
        while (!ready.empty()) {
            std::string current = ready.front();
            ready.pop();
            order.push_back(current);

            for (const auto& neighbor : adjacency.at(current)) {
                localInDegree[neighbor]--;
                if (localInDegree[neighbor] == 0) {
                    ready.push(neighbor);
                }
            }
        }

        if (order.size() != adjacency.size()) {
            std::cout << "Error: circular prerequisite dependency detected. "
                      << "Cannot generate a valid course order." << std::endl;
            return {};
        }

        return order;
    }

    size_t courseCount() const {
        return adjacency.size();
    }
};

// main.cpp
// ENHANCEMENT (Milestone Three — Algorithms and Data Structures).
//
// Two enhancements over the original artifact:
//  1. The plain BinarySearchTree has been replaced with a self-balancing
//     AVLTree, guaranteeing O(log n) search/insert regardless of load
//     order.
//  2. Prerequisites are now modeled as a directed graph (CourseGraph),
//     which can detect circular dependencies and generate a valid
//     course-taking order -- neither of which the original code could do.
#include "Course.hpp"
#include "AVLTree.hpp"
#include "HashTable.hpp"
#include "CourseGraph.hpp"
#include <iostream>
#include <fstream>
#include <sstream>
#include <vector>
#include <algorithm>

std::vector<Course> courseVector;
AVLTree courseTree;
HashTable courseHashTable;
CourseGraph prerequisiteGraph;

void loadCourses(const std::string& filename) {
    std::ifstream file(filename);
    if (!file.is_open()) {
        std::cout << "Error: File not found." << std::endl;
        return;
    }

    std::string line;
    while (std::getline(file, line)) {
        std::stringstream ss(line);
        std::string token;
        std::vector<std::string> tokens;
        while (std::getline(ss, token, ',')) {
            tokens.push_back(token);
        }
        if (tokens.size() < 2) continue;

        Course course;
        course.courseId = tokens[0];
        course.courseName = tokens[1];
        for (size_t i = 2; i < tokens.size(); i++) {
            course.prerequisites.push_back(tokens[i]);
        }

        courseVector.push_back(course);
        courseTree.insert(course); // now a self-balancing AVL insert
        courseHashTable.insert(course);

        prerequisiteGraph.addCourse(course.courseId);
        for (const auto& prereq : course.prerequisites) {
            prerequisiteGraph.addPrerequisiteEdge(prereq, course.courseId);
        }
    }

    std::cout << "Courses loaded successfully." << std::endl;
    std::cout << "AVL tree height after loading: " << courseTree.getHeight()
              << " for " << courseVector.size() << " courses "
              << "(a plain BST loading these same IDs in sorted order "
              << "would have a height of " << courseVector.size() << ")" << std::endl;
}

bool validateCourses() {
    for (const auto& course : courseVector) {
        for (const auto& prereq : course.prerequisites) {
            bool found = false;
            for (const auto& c : courseVector) {
                if (c.courseId == prereq) {
                    found = true;
                    break;
                }
            }
            if (!found) {
                std::cout << "ERROR: Prerequisite " << prereq
                          << " missing for " << course.courseId << std::endl;
                return false;
            }
        }
    }

    // ENHANCEMENT: also check for circular prerequisite chains using the
    // graph's topological sort -- the original version could not do this.
    std::vector<std::string> order = prerequisiteGraph.topologicalSort();
    if (order.empty() && !courseVector.empty()) {
        return false; // explanatory message already printed by topologicalSort()
    }

    std::cout << "All prerequisites are valid, and no circular dependencies were found." << std::endl;
    return true;
}

void searchCourse(const std::string& courseId) {
    for (const auto& course : courseVector) {
        if (course.courseId == courseId) {
            std::cout << course.courseId << " - " << course.courseName << std::endl;
            std::cout << "Prerequisites: ";
            if (course.prerequisites.empty()) {
                std::cout << "None";
            }
            for (size_t i = 0; i < course.prerequisites.size(); i++) {
                std::cout << course.prerequisites[i];
                if (i != course.prerequisites.size() - 1) std::cout << ", ";
            }
            std::cout << std::endl;
            return;
        }
    }
    std::cout << "Error: Course not found." << std::endl;
}

void printCoursesSorted() {
    std::vector<Course> sorted = courseVector;
    std::sort(sorted.begin(), sorted.end(), [](const Course& a, const Course& b) {
        return a.courseId < b.courseId;
    });
    for (const auto& c : sorted) {
        std::cout << c.courseId << " - " << c.courseName << std::endl;
    }
}

void printCourseSequence() {
    // ENHANCEMENT: brand-new feature. Generates a valid course-taking
    // order using a topological sort over the prerequisite graph.
    std::vector<std::string> order = prerequisiteGraph.topologicalSort();
    if (order.empty()) {
        return; // error already printed inside topologicalSort()
    }
    std::cout << "Recommended course order:" << std::endl;
    for (size_t i = 0; i < order.size(); i++) {
        std::cout << (i + 1) << ". " << order[i] << std::endl;
    }
}

        std::cout << "5. Generate Valid Course Sequence (new)\n";
        std::cout << "9. Exit\n";
        std::cout << "Choice: ";
        if (!(std::cin >> choice)) break;

        if (choice == 1) {
            loadCourses(filename);
        } else if (choice == 2) {

        } else if (choice == 3) {
